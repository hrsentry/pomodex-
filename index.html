function go(id){document.querySelectorAll('.view').forEach(v=>v.classList.toggle('active',v.id===id));if(id==='dex')renderClocks();if(id==='customise')renderPalettes();if(id==='clock')renderClock();if(id==='editor')loadEditor();window.scrollTo({top:0,behavior:'smooth'});}

function login(e){e.preventDefault();let user=data.user;if(!user||user.username.toLowerCase()!==$('loginUser').value.trim().toLowerCase()||user.password!==$('loginPass').value){alert('No matching local profile was found. Create a profile in this browser first.');return}alert('Welcome back, '+user.username+'!');updateAccountNav();go('dex')}

function logout(){data.user=null;persist();updateAccountNav();alert('You have logged out of this local demo.');go('home')}

function renderClocks(){let el=$('clockList');if(!el)return;el.innerHTML='';data.clocks.forEach(c=>{let item=document.createElement('div');item.className='clock-item';item.innerHTML=`<div><strong>${escapeHtml(c.name)}</strong><p>${c.focus} min focus · ${c.short} min break</p></div><div style="display:flex;gap:6px"><button class="btn small" data-open="${c.id}">View</button><button class="btn small" data-edit="${c.id}">Edit</button></div>`;el.append(item)});el.querySelectorAll('[data-open]').forEach(b=>b.onclick=()=>{selectClock(+b.dataset.open);go('clock')});el.querySelectorAll('[data-edit]').forEach(b=>b.onclick=()=>{selectClock(+b.dataset.edit);go('editor')});}

function addClock(){let id=Math.max(0,...data.clocks.map(c=>c.id))+1;data.clocks.push({id,name:'Clock #'+id,focus:25,short:5,long:15,cycles:4,quote:'One thing at a time.',description:'',palette:data.palette,lastUsed:'Never'});selectClock(id);persist();renderClocks();go('editor')}

function applyCustomPreview(){let cs=[...$('colorInputs').querySelectorAll('input')].map(i=>i.value);cs.forEach((c,i)=>document.documentElement.style.setProperty(['--c1','--c2','--c3','--c4','--c5'][i],c))}

function savePalette(){let name=$('paletteName').value.trim()||'My palette',colors=[...$('colorInputs').querySelectorAll('input')].map(i=>i.value);let existing=data.palettes.find(p=>p.name.toLowerCase()===name.toLowerCase());if(existing)existing.colors=colors;else data.palettes.push({name,colors});data.palette=name;data.clocks.forEach(c=>c.palette=name);applyPalette();persist();renderPalettes();$('paletteNotice').textContent='“'+name+'” has been saved and applied.'}

function resetCustom(){let p=data.palettes.find(p=>p.name===data.palette)||PRESETS[0];$('colorInputs').querySelectorAll('input').forEach((i,n)=>i.value=p.colors[n]);applyCustomPreview()}
